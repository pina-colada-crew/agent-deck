## Overview

Implement drag-and-drop functionality for sessions and session groups in the Paragon Agent Manager client, enabling users to:
1. Drag sessions between groups and reorder within groups
2. Drag groups into other groups (re-nesting) with circular dependency prevention
3. Auto-expand collapsed groups during hover (1-second delay)

- **No dnd-kit installed** - greenfield implementation
- **SessionList.tsx** renders groups and sessions hierarchically via `buildGroupTree()`

## Desired End State

After implementation:
1. Users can drag sessions to reorder within a group
2. Users can drag sessions between groups (including to/from "Misc.")
3. Users can drag groups into other groups (with circular dependency prevention)
4. Collapsed groups auto-expand after 1-second hover during drag
5. Visual feedback shows valid/invalid drop targets
6. Session order within groups persists across app restarts
7. All operations sync via WebSocket for real-time updates

### Verification
- Drag a session within its group → order changes and persists after refresh
- Drag a session to another group → session appears in new group
- Drag a group into another group → group becomes nested child
- Cannot drag a group into itself or its descendants (validation error)
- Hover over collapsed group during drag → expands after 1 second

## What We're NOT Doing

- Keyboard-based reordering (K/J hotkeys exist in TUI, not in web client)
- Multi-select drag (drag one item at a time)
- Cross-window drag-and-drop
- Undo/redo for drag operations
- Animation/transition effects beyond basic DragOverlay

## Implementation Approach

Use `@dnd-kit` library following patterns from the research document. Key architectural decisions:

1. **Two-layer context pattern**: `SessionDragStateContext` (raw state) + `SessionDragContext` (validation logic)
2. **Unified draggable types**: `type: "session" | "group"` to distinguish items
3. **Path-based identification**: Items identified by session ID or group path
4. **Optimistic updates**: Update UI immediately, revert on API failure
5. **RTK Query cache invalidation**: Invalidate `Sessions` and `Groups` tags after drag operations

```typescript
import { createContext, useContext, useState, useCallback, ReactNode } from "react";
import type { Active, Over } from "@dnd-kit/core";

// Types for draggable items
export type DraggableType = "session" | "group";

export interface DragItem {
  id: string;
  type: DraggableType;
  name: string;
  groupPath?: string; // For sessions: current group; For groups: the group path
}

// Raw state context
interface SessionDragStateContextType {
  activeItem: DragItem | null;
  setActiveItem: (item: DragItem | null) => void;
  overTarget: string | null;
  setOverTarget: (id: string | null) => void;
}

const SessionDragStateContext = createContext<SessionDragStateContextType | null>(null);

// Business logic context
interface SessionDragContextType {
  activeItem: DragItem | null;
  overTarget: string | null;
  setOverTarget: (id: string | null) => void;
  isDroppable: (targetPath: string, targetType: DraggableType) => boolean;
}

const SessionDragContext = createContext<SessionDragContextType | null>(null);

// Provider component
interface SessionDragProviderProps {
  children: ReactNode;
  groups: { path: string; name: string }[];
}

export function SessionDragProvider({ children, groups }: SessionDragProviderProps) {
  const [activeItem, setActiveItem] = useState<DragItem | null>(null);
  const [overTarget, setOverTarget] = useState<string | null>(null);

  // Validation: Can we drop the active item on this target?
  const isDroppable = useCallback((targetPath: string, targetType: DraggableType): boolean => {
    if (!activeItem) return false;

    // Sessions can be dropped on any group (including root/ungrouped)
    if (activeItem.type === "session") {
      // Can't drop on self (no-op)
      if (targetPath === activeItem.groupPath) return false;
      return true;
    }

    // Groups have more restrictions
    if (activeItem.type === "group") {
      // Can't drop group on a session
      if (targetType === "session") return false;

      // Can't drop group into itself
      if (targetPath === activeItem.id) return false;

      // Can't drop group into its own descendant (circular)
      if (targetPath.startsWith(activeItem.id + "/")) return false;

      // Can't drop group into its current parent (no-op)
      const currentParent = activeItem.id.includes("/")
        ? activeItem.id.substring(0, activeItem.id.lastIndexOf("/"))
        : "";
      if (targetPath === currentParent) return false;

      return true;
    }

    return false;
  }, [activeItem]);

  const stateValue: SessionDragStateContextType = {
    activeItem,
    setActiveItem,
    overTarget,
    setOverTarget,
  };

  const contextValue: SessionDragContextType = {
    activeItem,
    overTarget,
    setOverTarget,
    isDroppable,
  };

  return (
    <SessionDragStateContext.Provider value={stateValue}>
      <SessionDragContext.Provider value={contextValue}>
        {children}
      </SessionDragContext.Provider>
    </SessionDragStateContext.Provider>
  );
}

// Hooks
export function useSessionDragState() {
  const context = useContext(SessionDragStateContext);
  if (!context) throw new Error("useSessionDragState must be used within SessionDragProvider");
  return context;
}

export function useSessionDrag() {
  const context = useContext(SessionDragContext);
  if (!context) throw new Error("useSessionDrag must be used within SessionDragProvider");
  return context;
}
```

// ReorderSessionsRequest is the JSON request for reordering sessions within a group
type ReorderSessionsRequest struct {
	GroupPath  string   `json:"group_path"`
	SessionIDs []string `json:"session_ids"` // Ordered list of session IDs
}

func (s *Server) handleReorderSessions(w http.ResponseWriter, r *http.Request) {
	var req ReorderSessionsRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}

	s.sessionsMu.Lock()
	defer s.sessionsMu.Unlock()

	// Update positions for all sessions in the request
	for i, sessionID := range req.SessionIDs {
		inst := s.findSession(sessionID)
		if inst == nil {
			continue
		}
		// Only update if session is in the specified group
		if inst.GroupPath == req.GroupPath || (inst.GroupPath == "" && req.GroupPath == "") {
			inst.Position = i
		}
	}

	if err := s.saveState(); err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	w.WriteHeader(http.StatusNoContent)
}
```

// MoveGroupRequest is the JSON request for moving a group to a new parent
type MoveGroupRequest struct {
	NewParentPath string `json:"new_parent_path"` // Empty string or "/" for root
}

func (s *Server) handleMoveGroup(w http.ResponseWriter, r *http.Request) {
	pathParam := chi.URLParam(r, "path")
	groupPath, err := url.PathUnescape(pathParam)
	if err != nil {
		http.Error(w, "Invalid group path", http.StatusBadRequest)
		return
	}

	var req MoveGroupRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}

	s.sessionsMu.Lock()
	defer s.sessionsMu.Unlock()

	errMsg := s.groupTree.MoveGroupToParent(groupPath, req.NewParentPath)
	if errMsg != "" {
		http.Error(w, errMsg, http.StatusBadRequest)
		return
	}

	if err := s.saveState(); err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	// Return updated group info
	// Find the group at its new path
	leafName := groupPath
	if idx := strings.LastIndex(groupPath, "/"); idx != -1 {
		leafName = groupPath[idx+1:]
	}
	newPath := leafName
	if req.NewParentPath != "" && req.NewParentPath != "/" {
		newPath = req.NewParentPath + "/" + leafName
	}

	group := s.groupTree.Groups[newPath]
	if group == nil {
		http.Error(w, "Group moved but not found at new location", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(GroupResponse{
		Name:         group.Name,
		Path:         group.Path,
		Expanded:     group.Expanded,
		Order:        group.Order,
		DefaultPath:  group.DefaultPath,
		SessionCount: len(group.Sessions),
	})
}
```

```typescript
import { useCallback } from "react";
import type { DragStartEvent, DragEndEvent, DragCancelEvent } from "@dnd-kit/core";
import { useSessionDragState, DragItem } from "../contexts/SessionDragContext";
import { useUpdateSessionMutation, useReorderSessionsMutation } from "../api/sessionsApi";
import { useMoveGroupMutation } from "../api/groupsApi";
import type { Session } from "../types/session";

interface UseDragHandlersProps {
  sessions: Session[];
  groups: { path: string; name: string }[];
}

export function useDragHandlers({ sessions, groups }: UseDragHandlersProps) {
  const { setActiveItem, setOverTarget } = useSessionDragState();
  const [updateSession] = useUpdateSessionMutation();
  const [reorderSessions] = useReorderSessionsMutation();
  const [moveGroup] = useMoveGroupMutation();

  const handleDragStart = useCallback((event: DragStartEvent) => {
    const { active } = event;
    const data = active.data.current as DragItem | undefined;

    if (data) {
      setActiveItem(data);
    }
  }, [setActiveItem]);

  const handleDragEnd = useCallback(async (event: DragEndEvent) => {
    const { active, over } = event;

    setActiveItem(null);
    setOverTarget(null);

    if (!over || active.id === over.id) {
      return;
    }

    const draggedItem = active.data.current as DragItem;
    const dropTarget = over.data.current as { type: "session" | "group" | "root"; path?: string } | undefined;

    if (!draggedItem || !dropTarget) return;

    // Handle session drag
    if (draggedItem.type === "session") {
      const targetGroupPath = dropTarget.type === "root" ? "" : (dropTarget.path || "");

      // Check if this is a reorder within the same group
      if (targetGroupPath === draggedItem.groupPath && dropTarget.type === "session") {
        // Reorder within group
        const groupSessions = sessions
          .filter(s => (s.group_path || "") === targetGroupPath)
          .sort((a, b) => (a.position || 0) - (b.position || 0));

        const oldIndex = groupSessions.findIndex(s => s.id === draggedItem.id);
        const newIndex = groupSessions.findIndex(s => s.id === over.id);

        if (oldIndex !== -1 && newIndex !== -1 && oldIndex !== newIndex) {
          // Create new order
          const newOrder = [...groupSessions];
          const [removed] = newOrder.splice(oldIndex, 1);
          newOrder.splice(newIndex, 0, removed);

          await reorderSessions({
            groupPath: targetGroupPath,
            sessionIds: newOrder.map(s => s.id),
          });
        }
      } else {
        // Move to different group
        await updateSession({
          id: draggedItem.id,
          group_path: targetGroupPath,
        });
      }
    }

    // Handle group drag
    if (draggedItem.type === "group") {
      const targetParentPath = dropTarget.type === "root" ? "" : (dropTarget.path || "");

      await moveGroup({
        path: draggedItem.id,
        newParentPath: targetParentPath,
      });
    }
  }, [sessions, setActiveItem, setOverTarget, updateSession, reorderSessions, moveGroup]);

  const handleDragCancel = useCallback((event: DragCancelEvent) => {
    setActiveItem(null);
    setOverTarget(null);
  }, [setActiveItem, setOverTarget]);

  return {
    handleDragStart,
    handleDragEnd,
    handleDragCancel,
  };
}
```

import { useEffect, useRef } from "react";

function GroupRow({ /* ... */ }: GroupRowProps) {
  // ... existing code ...

  const expandTimerRef = useRef<NodeJS.Timeout | null>(null);
  const { activeItem } = useSessionDrag();

  // Auto-expand on hover during drag
  useEffect(() => {
    // Only expand if:
    // 1. We're hovering over this group
    // 2. A drag is in progress
    // 3. The group is collapsed
    // 4. The drop would be valid
    if (isOver && activeItem && isCollapsed && canDrop) {
      expandTimerRef.current = setTimeout(() => {
        toggleGroup(node.path);
      }, 1000); // 1 second delay
    }

    return () => {
      if (expandTimerRef.current) {
        clearTimeout(expandTimerRef.current);
        expandTimerRef.current = null;
      }
    };
  }, [isOver, activeItem, isCollapsed, canDrop, toggleGroup, node.path]);

  // ... rest of component
}
